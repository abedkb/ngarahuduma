-- Gawa madeni yenye bidhaa za matawi mengi kuwa deni MOJA kwa kila tawi, ili kila tawi lione deni lake tu. -- Walioathirika (kulingana na data yako): GABO KANAZI, MR RUTHA DODOMA, YANDAN BENJAMIN. -- Malipo yaliyokwisha kufanywa yanagawiwa kwa mtindo wa "kongwe kwanza" (bidhaa za tarehe za mwanzo zimelipwa kwanza); -- jumla ya deni, malipo na salio kwa mdaiwa haibadiliki. Nakala ya kila kitu inahifadhiwa kwenye hifadhi_nakala. -- Endesha BLOCK 1 (inaonyesha tu), kisha BLOCK 2. Hii inachukua nafasi ya ngara-huduma-madeni-3-assign-tawi.sql (usiiendeshe). -- ===== BLOCK 1 (preview) — seli moja ===== select coalesce(string_agg(w.jina||' (deni '||w.deni_lote||', amelipa '||w.kilicholipwa||'): '||t.maelezo, E'\n' order by w.jina),'hakuna wa kugawanywa') as matokeo from wateja w join (select v.mteja_id, string_agg(v.tawi||'='||v.s, ' | ' order by v.mn) as maelezo, count(*) as idadi_matawi from (select mteja_id, tawi, sum(jumla) s, min(tarehe) mn from madeni_binafsi_vitu where tawi is not null group by mteja_id, tawi) v group by v.mteja_id having count(*) > 1) t on t.mteja_id = w.id::text where w.binafsi and w.deleted_at is null; -- ===== BLOCK 2 (apply) ===== do $$ declare w record; b record; paid_left numeric; alloc numeric; main_t text; new_id uuid; n int := 0; tot numeric; begin create table if not exists hifadhi_nakala ( id bigserial primary key, jedwali text, nakala jsonb, imehifadhiwa timestamptz default now(), sababu text); for w in select x.* from wateja x where x.binafsi and x.deleted_at is null and (select count(distinct v.tawi) from madeni_binafsi_vitu v where v.mteja_id = x.id::text and v.tawi is not null) > 1 loop select sum(jumla) into tot from madeni_binafsi_vitu where mteja_id = w.id::text and tawi is not null; if abs(coalesce(tot,0) - coalesce(w.deni_lote,0)) > 1 then raise exception 'Jumla ya bidhaa za % (%) hailingani na deni (%) — sijagawa chochote', w.jina, tot, w.deni_lote; end if; insert into hifadhi_nakala(jedwali,nakala,sababu) values ('wateja', to_jsonb(w), 'gawa madeni kwa tawi'); insert into hifadhi_nakala(jedwali,nakala,sababu) select 'madeni_binafsi_vitu', to_jsonb(v), 'gawa madeni kwa tawi' from madeni_binafsi_vitu v where v.mteja_id = w.id::text; select tawi into main_t from (select tawi, sum(jumla) s from madeni_binafsi_vitu where mteja_id = w.id::text and tawi is not null group by tawi) q order by s desc, tawi limit 1; paid_left := coalesce(w.kilicholipwa, 0); for b in select tawi, sum(jumla) as deni, min(tarehe) as mn from madeni_binafsi_vitu where mteja_id = w.id::text and tawi is not null group by tawi order by min(tarehe), sum(jumla) desc, tawi loop alloc := least(paid_left, b.deni); paid_left := paid_left - alloc; if b.tawi = main_t then update wateja set tawi = b.tawi, deni_lote = b.deni, kilicholipwa = alloc, bakaa = b.deni - alloc, hali = case when b.deni - alloc <= 0 then 'Amelipa Kamili' when alloc > 0 then 'Amelipa Sehemu' else 'Hajalipa' end where id = w.id; else insert into wateja (jina, simu, tawi, aina_mteja, chanzo, deni_lote, kilicholipwa, bakaa, tarehe_mwisho, hali, binafsi) values (w.jina, w.simu, b.tawi, 'Kawaida', 'Bidhaa', b.deni, alloc, b.deni - alloc, w.tarehe_mwisho, case when b.deni - alloc <= 0 then 'Amelipa Kamili' when alloc > 0 then 'Amelipa Sehemu' else 'Hajalipa' end, true) returning id into new_id; update madeni_binafsi_vitu set mteja_id = new_id::text where mteja_id = w.id::text and tawi = b.tawi; end if; end loop; n := n + 1; end loop; raise notice 'Madeni yaliyogawanywa: %', n; end $$; -- ===== CHECK — seli moja ===== select 'bila tawi: ' || count(*) filter (where tawi is null) || ' | yenye tawi: ' || count(*) filter (where tawi is not null) || E'\n' || coalesce((select string_agg(jina||' | '||tawi||' | deni '||deni_lote||' | amelipa '||kilicholipwa||' | bakaa '||bakaa, E'\n' order by jina, tawi) from wateja where binafsi and deleted_at is null and upper(trim(jina)) in ('GABO KANAZI','MR RUTHA DODOMA','YANDAN BENJAMIN')), '') as matokeo from wateja where binafsi and deleted_at is null;
